cmake_minimum_required(VERSION 3.21)
project(videogallery_ffmpeg C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_SKIP_RPATH TRUE)
foreach(name avutil swresample avcodec)
    add_library(${name} SHARED IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${FFMPEG_PREFIX}/lib/lib${name}.so")
endforeach()
add_library(ffmpegJNI SHARED "${FFMPEG_JNI_SOURCE}")
target_include_directories(ffmpegJNI PRIVATE "${FFMPEG_PREFIX}/include")
set_target_properties(ffmpegJNI PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_link_libraries(ffmpegJNI PRIVATE android swresample avcodec avutil log)
